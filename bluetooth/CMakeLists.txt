cmake_minimum_required(VERSION 3.18)
project(airmouse_hid C)
set(CMAKE_C_STANDARD 11)
if(NOT EXISTS "${BTSTACK_ROOT}/src/hci.c")
  message(FATAL_ERROR "BTSTACK_ROOT must name the pinned BTstack source tree")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(BT_ROOT_SOURCES ad_parser btstack_crypto btstack_hid_parser btstack_linked_list btstack_memory btstack_memory_pool btstack_ring_buffer btstack_run_loop btstack_run_loop_base btstack_tlv btstack_util hci hci_cmd hci_event hci_event_builder hci_dump hci_transport_h4 l2cap l2cap_signaling)
set(BT_SOURCES)
foreach(name IN LISTS BT_ROOT_SOURCES)
  list(APPEND BT_SOURCES "${BTSTACK_ROOT}/src/${name}.c")
endforeach()
foreach(name att_db att_dispatch att_server gatt_client le_device_db_tlv sm)
  list(APPEND BT_SOURCES "${BTSTACK_ROOT}/src/ble/${name}.c")
endforeach()
foreach(name btstack_run_loop_posix btstack_tlv_posix)
  list(APPEND BT_SOURCES "${BTSTACK_ROOT}/platform/posix/${name}.c")
endforeach()
# The transport is built from upstream source with exclusive UART ownership.
file(READ "${BTSTACK_ROOT}/platform/posix/btstack_uart_posix.c" UART_SOURCE)
string(REPLACE "#include <termios.h>" "#include <termios.h>\n#include <sys/ioctl.h>" UART_SOURCE "${UART_SOURCE}")
set(UART_ANCHOR "    if (tcgetattr(fd, &btstack_uart_block_termios) < 0) {")
string(FIND "${UART_SOURCE}" "${UART_ANCHOR}" UART_POSITION)
if(UART_POSITION EQUAL -1)
  message(FATAL_ERROR "Pinned UART ownership anchor changed")
endif()
string(REPLACE "${UART_ANCHOR}" "    if (ioctl(fd, TIOCEXCL) < 0) { close(fd); return -1; }\n${UART_ANCHOR}" UART_SOURCE "${UART_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/btstack_uart_exclusive.c" "${UART_SOURCE}")
list(APPEND BT_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/btstack_uart_exclusive.c" "${BTSTACK_ROOT}/chipset/bcm/btstack_chipset_bcm.c" "${BTSTACK_ROOT}/src/ble/gatt-service/hids_device.c" "${BTSTACK_ROOT}/src/ble/gatt-service/battery_service_server.c" "${BTSTACK_ROOT}/3rd-party/micro-ecc/uECC.c" "${BTSTACK_ROOT}/3rd-party/rijndael/rijndael.c")
add_library(btstack STATIC ${BT_SOURCES})
target_include_directories(btstack PUBLIC . "${CMAKE_CURRENT_BINARY_DIR}" "${BTSTACK_ROOT}/src" "${BTSTACK_ROOT}/platform/posix" "${BTSTACK_ROOT}/chipset/bcm" "${BTSTACK_ROOT}/3rd-party/micro-ecc" "${BTSTACK_ROOT}/3rd-party/rijndael" "${BTSTACK_ROOT}/3rd-party/tinydir")
target_compile_options(btstack PRIVATE -ffunction-sections -fdata-sections)
target_link_libraries(btstack PUBLIC Threads::Threads)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/airmouse_gatt.h" COMMAND "${Python3_EXECUTABLE}" "${BTSTACK_ROOT}/tool/compile_gatt.py" "${CMAKE_CURRENT_SOURCE_DIR}/airmouse.gatt" "${CMAKE_CURRENT_BINARY_DIR}/airmouse_gatt.h" DEPENDS airmouse.gatt VERBATIM)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/compatible_gatt.h"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/schema/compatible.py" "${BTSTACK_ROOT}/tool/compile_gatt.py" "${CMAKE_CURRENT_BINARY_DIR}/airmouse_gatt.h" "${CMAKE_CURRENT_BINARY_DIR}/compatible_gatt.h" "${CMAKE_CURRENT_BINARY_DIR}/lg_gatt.h"
  DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/airmouse_gatt.h" "${CMAKE_CURRENT_BINARY_DIR}/lg_gatt.h" schema/compatible.py schema/mouse-v1.gatt schema/media-v2.gatt VERBATIM)
add_library(gatt-migration STATIC gatt_migration.c "${CMAKE_CURRENT_BINARY_DIR}/compatible_gatt.h")
target_link_libraries(gatt-migration PUBLIC btstack)
target_compile_options(gatt-migration PRIVATE -Wall -Wextra -Werror)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/lg_gatt.h" COMMAND "${Python3_EXECUTABLE}" "${BTSTACK_ROOT}/tool/compile_gatt.py" "${CMAKE_CURRENT_SOURCE_DIR}/lg.gatt" "${CMAKE_CURRENT_BINARY_DIR}/lg_gatt.h" DEPENDS lg.gatt VERBATIM)
add_library(lg-remote STATIC lg_remote.c lg_gatt.c "${CMAKE_CURRENT_BINARY_DIR}/lg_gatt.h")
target_include_directories(lg-remote PUBLIC . "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(lg-remote PRIVATE -Wall -Wextra -Werror)
add_executable(lg-remote-test lg_remote.c test/lg_remote_test.c)
target_include_directories(lg-remote-test PRIVATE .)
target_compile_options(lg-remote-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_executable(airmouse-hid airmouse_hid.c hid_state.c host_registry.c "${CMAKE_CURRENT_BINARY_DIR}/airmouse_gatt.h")
target_compile_options(airmouse-hid PRIVATE -Wall -Wextra -Werror)
target_link_options(airmouse-hid PRIVATE -Wl,--gc-sections)
target_link_libraries(airmouse-hid PRIVATE gatt-migration lg-remote btstack)
add_executable(state-test hid_state.c test/state_test.c)
target_include_directories(state-test PRIVATE .)
target_compile_options(state-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME lg-remote COMMAND lg-remote-test)
add_test(NAME hid-state COMMAND state-test)
add_executable(lifecycle-test hid_state.c host_registry.c test/lifecycle_test.c)
add_dependencies(lifecycle-test airmouse-hid)
target_compile_options(lifecycle-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_options(lifecycle-test PRIVATE -Wl,--gc-sections)
target_link_libraries(lifecycle-test PRIVATE gatt-migration lg-remote btstack)
add_test(NAME hid-lifecycle COMMAND lifecycle-test)
if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME hid-ipc COMMAND "${CMAKE_COMMAND}" -E env "AIRMOUSE_HID_BINARY=$<TARGET_FILE:airmouse-hid>" "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test/test_ipc.py")
endif()

add_executable(registry-test host_registry.c test/registry_test.c)
target_include_directories(registry-test PRIVATE .)
target_compile_options(registry-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_options(registry-test PRIVATE -Wl,--wrap=fsync,--wrap=rename)
add_test(NAME host-registry COMMAND registry-test)
add_executable(gatt-migration-test test/gatt_migration_test.c)
target_compile_options(gatt-migration-test PRIVATE -UNDEBUG -ffunction-sections -fdata-sections)
target_link_options(gatt-migration-test PRIVATE -Wl,--gc-sections)
target_link_libraries(gatt-migration-test PRIVATE gatt-migration lg-remote btstack)
add_test(NAME gatt-migration COMMAND gatt-migration-test)
add_test(NAME gatt-schema COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test/test_schema.py")
add_test(NAME lg-inspect COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test/test_lg_inspect.py")
